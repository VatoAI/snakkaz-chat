# EMERGENCY MIME TYPE FIX for LiteSpeed
# Multiple approaches to ensure JavaScript files serve correctly

# Method 1: AddType directives
<IfModule mod_mime.c>
    # Primary JavaScript MIME type
    AddType application/javascript .js
    AddType application/javascript .mjs
    AddType text/javascript .js
    
    # Other assets
    AddType text/css .css
    AddType application/json .json
    AddType image/svg+xml .svg
    AddType application/font-woff .woff
    AddType application/font-woff2 .woff2
    AddType application/vnd.ms-fontobject .eot
    AddType application/x-font-ttf .ttf
</IfModule>

# Method 2: Force headers with mod_headers (LiteSpeed compatible)
<IfModule mod_headers.c>
    # Force JavaScript MIME type for all .js files
    <FilesMatch "\.(js|mjs)$">
        Header unset Content-Type
        Header always set Content-Type "application/javascript; charset=utf-8"
        Header always set X-Content-Type-Options "nosniff"
        Header always set Cache-Control "public, max-age=31536000"
    </FilesMatch>
    
    # Force CSS MIME type
    <FilesMatch "\.css$">
        Header unset Content-Type
        Header always set Content-Type "text/css; charset=utf-8"
    </FilesMatch>
    
    # Specific override for assets directory
    <FilesMatch "^assets/.*\.(js|mjs)$">
        Header unset Content-Type
        Header always set Content-Type "application/javascript"
        Header always set Access-Control-Allow-Origin "*"
    </FilesMatch>
</IfModule>

# Method 3: Environment-based MIME type setting
SetEnvIf Request_URI "\.js$" js_file=1
SetEnvIf Request_URI "\.mjs$" js_file=1
Header always set Content-Type "application/javascript" env=js_file
